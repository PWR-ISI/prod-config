# Pobierz token (id_token) po zalogowaniu:
function Get-Token($email,$pass){
  $r = Invoke-RestMethod -Uri "$AUTH/auth/login/" -Method POST -ContentType 'application/json' `
       -Body (@{email=$email;password=$pass} | ConvertTo-Json)
  return $r.id_token
}
$ADMIN = Get-Token 'admin@isi.test' 'Demo1234!'
$PATIENT = Get-Token 'patient@isi.test' 'Demo1234!'

# Uniwersalny wywoływacz (pokazuje też ciało błędu np. 400/401/403):
function Api($method,$url,$body,$token){
  $h=@{}; if($token){$h['Authorization']="Bearer $token"}
  try { $r=Invoke-WebRequest -Uri $url -Method $method -Headers $h -Body $body -ContentType 'application/json' -UseBasicParsing -TimeoutSec 30
        "[{0}] {1}" -f [int]$r.StatusCode, $r.Content }
  catch { $e=$_.Exception.Response; if($e){$s=New-Object IO.StreamReader($e.GetResponseStream()); "[{0}] {1}" -f [int]$e.StatusCode, $s.ReadToEnd()} else {"[ERR] $($_.Exception.Message)"} }
}
